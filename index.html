Busca de perfis no Bluesky

Encontre perfis por um termo que eles postaram, por posts recentes em português ou por texto na bio, veja os seguidores de alguém, ou descubra quem você segue e não te segue de volta. Dá para filtrar por número de seguidores e por atividade recente, e seguir ou deixar de seguir direto da lista.

Conectado como

Entre com uma app password, nunca a senha da sua conta. Você cria uma em Configurações → App Passwords no Bluesky, e pode revogá-la quando quiser — ela não dá acesso a e-mail, senha nem à exclusão do perfil. Esta página não tem servidor: sua senha vai direto para o Bluesky e só os tokens da sessão ficam guardados neste navegador.

Script para seguir todos os listados

A página já tem o botão Seguir todos os listados. Se preferir usar o console do navegador (F12 → Console), copie este script. Ele espera cada follow terminar, registra sucessos e falhas e faz uma pausa aleatória de 10 a 30 segundos entre as tentativas. 

async function clicarEmTodosSeguir() {
  const botoes = document.querySelectorAll(
    'button.seguir[data-did^="did:plc:"]'
  );

  let feitos = 0;
  let erros = 0;

  for (const [indice, botao] of botoes.entries()) {
    botao.dataset.resultado = '';
    botao.click();

    const resultado = await new Promise(resolve => {
      const limite = Date.now() + 60000;
      const conferir = () => {
        if (botao.dataset.resultado) return resolve(botao.dataset.resultado);
        if (Date.now() >= limite) return resolve('erro');
        setTimeout(conferir, 250);
      };
      conferir();
    });

    if (resultado === 'ok') feitos++;
    else erros++;

    const tempoEspera = Math.floor(Math.random() * (30000 - 10000 + 1)) + 10000;

    console.log(
      `${indice + 1}/${botoes.length}: ${resultado === 'ok' ? 'seguido' : 'falhou'}.` +
      (indice < botoes.length - 1 ? ` Próximo em ${(tempoEspera / 1000).toFixed(1)} segundos.` : '')
    );

    if (indice < botoes.length - 1) {
      await new Promise(resolve => setTimeout(resolve, tempoEspera));
    }
  }

  console.log(`Finalizado: ${feitos} seguidos, ${erros} falharam.`);
}

clicarEmTodosSeguir();
